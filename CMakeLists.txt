cmake_minimum_required(VERSION 3.20)
project(PaperNX VERSION 0.9.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra)
endif()

add_library(papernx_core STATIC src/core.cpp src/gestures.cpp src/cover_repository.cpp src/catalog.cpp src/reading_stats.cpp src/reading_journal.cpp)
target_include_directories(papernx_core PUBLIC include)
include(CTest)
function(enable_test_assertions target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endfunction()
if(BUILD_TESTING)
    add_executable(papernx_catalog_tests tests/catalog_tests.cpp)
    enable_test_assertions(papernx_catalog_tests)
    target_link_libraries(papernx_catalog_tests PRIVATE papernx_core)
    add_test(NAME papernx_catalog_tests COMMAND papernx_catalog_tests)
    add_executable(papernx_foundations_tests tests/foundations_tests.cpp)
    enable_test_assertions(papernx_foundations_tests)
    target_link_libraries(papernx_foundations_tests PRIVATE papernx_core)
    add_test(NAME papernx_foundations_tests COMMAND papernx_foundations_tests)
    add_executable(papernx_localization_tests tests/localization_tests.cpp)
    enable_test_assertions(papernx_localization_tests)
    target_link_libraries(papernx_localization_tests PRIVATE papernx_core)
    add_test(NAME papernx_localization_tests COMMAND papernx_localization_tests)
    add_executable(papernx_cover_tests tests/cover_repository_tests.cpp)
    enable_test_assertions(papernx_cover_tests)
    target_link_libraries(papernx_cover_tests PRIVATE papernx_core)
    add_test(NAME papernx_cover_tests COMMAND papernx_cover_tests)
    add_executable(papernx_core_tests tests/core_tests.cpp)
    enable_test_assertions(papernx_core_tests)
    target_link_libraries(papernx_core_tests PRIVATE papernx_core)
    add_test(NAME papernx_core_tests COMMAND papernx_core_tests)
    add_executable(papernx_reading_journal_tests tests/reading_journal_tests.cpp)
    enable_test_assertions(papernx_reading_journal_tests)
    target_link_libraries(papernx_reading_journal_tests PRIVATE papernx_core)
    add_test(NAME papernx_reading_journal_tests COMMAND papernx_reading_journal_tests)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME papernx_utf8_integrity COMMAND ${Python3_EXECUTABLE} tests/check_utf8_integrity.py)
    set_tests_properties(papernx_utf8_integrity PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
endif()

option(PAPERNX_NATIVE "Build SDL2/MuPDF native reader (dependencies required)" OFF)
if(PAPERNX_NATIVE)
    find_package(Threads REQUIRED)
    find_package(PkgConfig QUIET)
    add_library(papernx_sdl INTERFACE)
    if(PKG_CONFIG_FOUND)
        pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
        pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
        target_link_libraries(papernx_sdl INTERFACE PkgConfig::SDL2 PkgConfig::SDL2_TTF)
    else()
        find_package(SDL2 REQUIRED)
        find_package(SDL2_ttf REQUIRED)
        target_include_directories(papernx_sdl INTERFACE ${SDL2_INCLUDE_DIRS} ${SDL2_TTF_INCLUDE_DIRS})
        target_link_libraries(papernx_sdl INTERFACE ${SDL2_LIBRARIES} ${SDL2_TTF_LIBRARIES})
    endif()

    if(MUPDF_PREFIX)
        # Never mix pinned headers/archives with an installed system version.
        find_path(MUPDF_INCLUDE_DIR mupdf/fitz.h PATHS "${MUPDF_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
        find_library(MUPDF_LIBRARY mupdf PATHS "${MUPDF_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
        find_library(MUPDF_THIRD_LIBRARY mupdf-third PATHS "${MUPDF_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
        pkg_check_modules(FONT_DEPS REQUIRED IMPORTED_TARGET harfbuzz freetype2)
        set(MUPDF_TRANSITIVE PkgConfig::FONT_DEPS)
    else()
        find_path(MUPDF_INCLUDE_DIR mupdf/fitz.h REQUIRED)
        find_library(MUPDF_LIBRARY mupdf REQUIRED)
        find_library(MUPDF_THIRD_LIBRARY mupdf-third)
        if(PKG_CONFIG_FOUND)
            pkg_check_modules(MUPDF_DEPS REQUIRED mupdf)
            pkg_check_modules(HARFBUZZ REQUIRED IMPORTED_TARGET harfbuzz)
            set(MUPDF_TRANSITIVE ${MUPDF_DEPS_STATIC_LIBRARIES} PkgConfig::HARFBUZZ)
        endif()
    endif()

    add_library(papernx_document STATIC src/mupdf_bridge.c)
    target_include_directories(papernx_document PUBLIC include PRIVATE ${MUPDF_INCLUDE_DIR})
    target_link_libraries(papernx_document PUBLIC ${MUPDF_LIBRARY})
    if(MUPDF_THIRD_LIBRARY)
        target_link_libraries(papernx_document PUBLIC ${MUPDF_THIRD_LIBRARY})
    endif()
    target_link_libraries(papernx_document PUBLIC ${MUPDF_TRANSITIVE} Threads::Threads)
    if(UNIX)
        target_link_libraries(papernx_document PUBLIC m ${CMAKE_DL_LIBS})
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(papernx_document PRIVATE -Werror=implicit-function-declaration)
    endif()
    add_library(papernx_ui STATIC src/typography.cpp src/page_cache.cpp src/reading_journal_ui.cpp)
    target_include_directories(papernx_ui PUBLIC include)
    target_link_libraries(papernx_ui PUBLIC papernx_sdl papernx_core)
    add_executable(PaperNX src/main.cpp)
    target_link_libraries(PaperNX PRIVATE papernx_ui papernx_document)
    if(BUILD_TESTING)
        add_executable(papernx_ui_component_tests tests/ui_component_tests.cpp)
        enable_test_assertions(papernx_ui_component_tests)
        target_link_libraries(papernx_ui_component_tests PRIVATE papernx_ui)
        add_test(NAME papernx_ui_component_tests COMMAND papernx_ui_component_tests)
        set_tests_properties(papernx_ui_component_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" ENVIRONMENT "SDL_VIDEODRIVER=dummy")
        add_executable(mupdf_render_tests tests/mupdf_render_tests.c)
        set_target_properties(mupdf_render_tests PROPERTIES LINKER_LANGUAGE CXX)
        enable_test_assertions(mupdf_render_tests)
        target_link_libraries(mupdf_render_tests PRIVATE papernx_document)
        add_test(NAME generate_cbz_fixture COMMAND ${Python3_EXECUTABLE} tests/generate_cbz_fixture.py)
        set_tests_properties(generate_cbz_fixture PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" FIXTURES_SETUP cbz_sample)
        add_test(NAME mupdf_render_tests COMMAND mupdf_render_tests)
        set_tests_properties(mupdf_render_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" FIXTURES_REQUIRED cbz_sample)
        add_executable(papernx_app_tests tests/app_tests.cpp)
        enable_test_assertions(papernx_app_tests)
        target_link_libraries(papernx_app_tests PRIVATE papernx_ui papernx_document)
        add_test(NAME papernx_app_tests COMMAND papernx_app_tests)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME v03_stress_fixtures COMMAND ${Python3_EXECUTABLE} tests/generate_stress_fixtures.py)
        set_tests_properties(v03_stress_fixtures PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" FIXTURES_SETUP v03_stress)
        set_tests_properties(papernx_app_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" ENVIRONMENT "SDL_VIDEODRIVER=dummy")
        set_tests_properties(papernx_app_tests PROPERTIES FIXTURES_REQUIRED v03_stress)
    endif()
endif()
