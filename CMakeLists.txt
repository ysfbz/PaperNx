option(PAPERNX_NATIVE "Build SDL2/MuPDF native reader (dependencies required)" OFF)
if(PAPERNX_NATIVE)
    find_package(Threads REQUIRED)
    file(GLOB_RECURSE PAPERNX_IDENTITY_INPUTS CONFIGURE_DEPENDS LIST_DIRECTORIES false
        "${CMAKE_SOURCE_DIR}/src/*" "${CMAKE_SOURCE_DIR}/include/*" "${CMAKE_SOURCE_DIR}/third_party/*"
        "${CMAKE_SOURCE_DIR}/assets/*" "${CMAKE_SOURCE_DIR}/fonts/*" "${CMAKE_SOURCE_DIR}/romfs/*")
    set(PAPERNX_IDENTITY_BUILD_INPUTS
        "${CMAKE_SOURCE_DIR}/Makefile" "${CMAKE_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_SOURCE_DIR}/Dockerfile"
        "${CMAKE_SOURCE_DIR}/scripts/build-mupdf-switch.sh" "${CMAKE_SOURCE_DIR}/scripts/artifact_validation.py"
        "${CMAKE_SOURCE_DIR}/scripts/build_receipt.py" "${CMAKE_SOURCE_DIR}/scripts/source_identity.py")
    list(APPEND PAPERNX_IDENTITY_INPUTS ${PAPERNX_IDENTITY_BUILD_INPUTS})
    set(PAPERNX_IDENTITY_CONFIGURE_INPUTS ${PAPERNX_IDENTITY_INPUTS})
    if(EXISTS "${CMAKE_SOURCE_DIR}/.git/HEAD" AND EXISTS "${CMAKE_SOURCE_DIR}/.git/index")
        list(APPEND PAPERNX_IDENTITY_CONFIGURE_INPUTS "${CMAKE_SOURCE_DIR}/.git/HEAD" "${CMAKE_SOURCE_DIR}/.git/index")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PAPERNX_IDENTITY_CONFIGURE_INPUTS})
    list(SORT PAPERNX_IDENTITY_INPUTS)
    string(ASCII 9 PAPERNX_TAB)
    set(PAPERNX_IDENTITY_MANIFEST "")
    foreach(PAPERNX_IDENTITY_FILE IN LISTS PAPERNX_IDENTITY_INPUTS)
        if(EXISTS "${PAPERNX_IDENTITY_FILE}")
            file(RELATIVE_PATH PAPERNX_IDENTITY_NAME "${CMAKE_SOURCE_DIR}" "${PAPERNX_IDENTITY_FILE}")
            file(SHA256 "${PAPERNX_IDENTITY_FILE}" PAPERNX_IDENTITY_FILE_SHA256)
            string(APPEND PAPERNX_IDENTITY_MANIFEST "${PAPERNX_IDENTITY_NAME}${PAPERNX_TAB}${PAPERNX_IDENTITY_FILE_SHA256}\n")
        endif()
    endforeach()
    string(SHA256 PAPERNX_SOURCE_TREE_SHA256 "${PAPERNX_IDENTITY_MANIFEST}")
    set(PAPERNX_SOURCE_COMMIT unknown)
    set(PAPERNX_SOURCE_DIRTY unknown)
    find_program(PAPERNX_GIT_EXECUTABLE git)
    if(PAPERNX_GIT_EXECUTABLE)
        execute_process(COMMAND "${PAPERNX_GIT_EXECUTABLE}" -c "safe.directory=${CMAKE_SOURCE_DIR}" rev-parse --verify HEAD
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" RESULT_VARIABLE PAPERNX_GIT_RESULT
            OUTPUT_VARIABLE PAPERNX_GIT_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        string(LENGTH "${PAPERNX_GIT_REVISION}" PAPERNX_GIT_REVISION_LENGTH)
        if(PAPERNX_GIT_RESULT EQUAL 0 AND PAPERNX_GIT_REVISION MATCHES "^[0-9a-fA-F]+$" AND
           (PAPERNX_GIT_REVISION_LENGTH EQUAL 40 OR PAPERNX_GIT_REVISION_LENGTH EQUAL 64))
            string(TOLOWER "${PAPERNX_GIT_REVISION}" PAPERNX_SOURCE_COMMIT)
            execute_process(COMMAND "${PAPERNX_GIT_EXECUTABLE}" -c "safe.directory=${CMAKE_SOURCE_DIR}" status --porcelain=v1 --untracked-files=normal
                WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" RESULT_VARIABLE PAPERNX_STATUS_RESULT
                OUTPUT_VARIABLE PAPERNX_GIT_STATUS ERROR_QUIET)
            if(PAPERNX_STATUS_RESULT EQUAL 0)
                string(STRIP "${PAPERNX_GIT_STATUS}" PAPERNX_GIT_STATUS)
                if(PAPERNX_GIT_STATUS STREQUAL "")
                    set(PAPERNX_SOURCE_DIRTY false)
                else()
                    set(PAPERNX_SOURCE_DIRTY true)
                endif()
            endif()
        endif()
    endif()
    if(PAPERNX_SOURCE_COMMIT STREQUAL "unknown")
        set(PAPERNX_SOURCE_IDENTITY "archive-sha256:${PAPERNX_SOURCE_TREE_SHA256}")
    else()
        set(PAPERNX_SOURCE_IDENTITY "git:${PAPERNX_SOURCE_COMMIT}:dirty:${PAPERNX_SOURCE_DIRTY}:tree:${PAPERNX_SOURCE_TREE_SHA256}")
    endif()
    add_compile_definitions(
        "PAPERNX_SOURCE_COMMIT=\"${PAPERNX_SOURCE_COMMIT}\""
        "PAPERNX_SOURCE_DIRTY=\"${PAPERNX_SOURCE_DIRTY}\""
        "PAPERNX_SOURCE_TREE_SHA256=\"${PAPERNX_SOURCE_TREE_SHA256}\""
        "PAPERNX_SOURCE_IDENTITY=\"${PAPERNX_SOURCE_IDENTITY}\"")

    endif()
